cmake_minimum_required(VERSION 3.16)
project(so101_planning_accel)
find_package(ament_cmake REQUIRED)
find_package(rosidl_default_generators REQUIRED)
find_package(rclcpp REQUIRED)
find_package(moveit_msgs REQUIRED)
find_package(moveit_core REQUIRED)
find_package(moveit_ros_planning REQUIRED)
rosidl_generate_interfaces(${PROJECT_NAME} "srv/PlanSnapshot.srv" "srv/CheckStates.srv" DEPENDENCIES moveit_msgs)
add_executable(planning_server src/planning_server.cpp)
target_compile_features(planning_server PUBLIC cxx_std_17)
ament_target_dependencies(planning_server rclcpp moveit_msgs moveit_core moveit_ros_planning)
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(planning_server "${cpp_typesupport_target}")
install(TARGETS planning_server DESTINATION lib/${PROJECT_NAME})
ament_export_dependencies(rosidl_default_runtime)
ament_package()
